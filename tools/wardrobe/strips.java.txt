    public static boolean wraps(int ci, int slot, int i) {
        return slot == 1 && ci >= 0 && ci < 8 && i >= 0 && i < PER && WRAPS[ci].charAt(i) == '1';
    }

    private static float wb(int ci, int i, int k) { return (WBS[ci].charAt(i * 3 + k) - 0x8000) / 1000f; }

    private static int fit(int ci, int idx, int row, int side) { return FITS[ci].charAt((idx * 40 + row) * 2 + side) - 0x100; }

    /** 몸 윤곽의 한 줄: y(칸 픽셀)에서 몸의 왼쪽·오른쪽 끝 → o. 몸이 없으면 false. 줄 사이는 이어 준다. */
    private static boolean bodyAt(int ci, int idx, float ty, float bh, float y, float[] o) {
        float t = (y - ty) / (0.025f * bh);
        int j = (int) Math.floor(t);
        float fr = t - j;
        int ja = Math.min(Math.max(j, 0), 39), jb = Math.min(Math.max(j + 1, 0), 39);
        int la = fit(ci, idx, ja, 0), ra = fit(ci, idx, ja, 1), lb = fit(ci, idx, jb, 0), rb = fit(ci, idx, jb, 1);
        boolean a = ra - la >= 8, b = rb - lb >= 8;
        if (a && b) { o[0] = la * (1 - fr) + lb * fr; o[1] = ra * (1 - fr) + rb * fr; return true; }
        if (a) { o[0] = la; o[1] = ra; return true; }
        if (b) { o[0] = lb; o[1] = rb; return true; }
        return false;
    }

    /**
     * 칸에 그려진 친구에게 입힐 그림 조각들. 옷은 몸 윤곽을 따라 줄(조각)마다 몸 폭에 맞춰 늘이고 줄여서 몸에 두른 것처럼 만들고,
     * 모자·소지품·두르지 않는 옷(목도리 · 나비넥타이 등)은 한 장 그대로. place() 로 구한 (x, y, u) 를 넣는다.
     * out[8*n..] = {그림 칸 안 원본 x0,y0,x1,y1 (192 기준 픽셀), 칸에 놓을 x0,y0,x1,y1 (칸 256 기준 픽셀)}. 반환 = 조각 수(0 = 못 그림).
     * out 길이는 8 * NS 이상.
     */
    public static int strips(int ci, int sheet, int cell, int slot, int i, float x, float y, float u, float[] out) {
        if (ci < 0 || ci >= 8 || slot < 0 || slot > 2 || i < 0 || i >= PER || out.length < 8 * NS) return 0;
        int b0 = (slot * PER + i) * 3;
        float fcx = WM[ci][b0], fcy = WM[ci][b0 + 1], side = WM[ci][b0 + 2];
        float sw = side * u;
        float cxs = x + fcx * u, cys = y + fcy * u, sx0 = cxs - sw / 2, sy0 = cys - sw / 2;
        float P = 192f / sw;
        int idx = sheet == 0 ? cell : sheet == 1 ? 16 + cell : 24 + cell;
        float[] bo = new float[2];
        float by0 = 0, by1 = 0, natw = 0, top = 0, bot = 0, ctr = 0, ty = 0, bh = 0;
        boolean wrap = wraps(ci, slot, i);
        if (wrap) {
            int[] m = MET[ci][sheet];
            int mb = cell * 5;
            if (cell < 0 || mb + 4 >= m.length) wrap = false;
            else {
                ty = m[mb + 1]; bh = m[mb + 3];
                by0 = wb(ci, i, 0); by1 = wb(ci, i, 1); natw = wb(ci, i, 2) * u;
                top = y + by0 * u; bot = y + by1 * u; ctr = (top + bot) / 2;
                if (natw <= 0 || !bodyAt(ci, idx, ty, bh, ctr, bo)) wrap = false;
            }
        }
        if (!wrap) {
            out[0] = 0; out[1] = 0; out[2] = 192; out[3] = 192;
            out[4] = sx0; out[5] = sy0; out[6] = sx0 + sw; out[7] = sy0 + sw;
            return 1;
        }
        float g = Math.min(Math.max((bo[1] - bo[0]) * 0.98f / natw, 1f), 1.5f);
        float H = (bot - top) * (float) Math.pow(g, 0.75);
        int n = 0;
        float lastS = -1, lastM = 0;
        String sx = SXS[ci];
        for (int k = 0; k < NS; k++) {
            float ya = ctr - H / 2 + k * H / NS, yb = ya + H / NS, yc = (ya + yb) / 2;
            int ql = sx.charAt((i * NS + k) * 2) - 0x100, qr = sx.charAt((i * NS + k) * 2 + 1) - 0x100;
            if (qr <= ql) continue;
            float sa = 96f + 192f * (by0 + (by1 - by0) * k / NS - fcy) / side, sb = 96f + 192f * (by0 + (by1 - by0) * (k + 1) / NS - fcy) / side;
            float nat = (qr - ql) / P, wd, mid;
            if (bodyAt(ci, idx, ty, bh, yc, bo)) {
                float bw = (bo[1] - bo[0]) * 0.98f;
                mid = (bo[0] + bo[1]) / 2;
                wd = Math.min(Math.max(bw, nat * g * 0.9f), nat * g * 1.15f);
                lastS = wd / nat; lastM = mid;
            } else if (lastS < 0) {
                wd = nat * g; mid = cxs + ((ql + qr) / 2f - 96f) / P;
            } else {
                wd = nat * lastS; mid = lastM;
            }
            int o = n * 8;
            int ia = Math.round(sa), ib = Math.round(sb);
            if (ib <= ia) ib = ia + 1;
            out[o] = ql; out[o + 1] = ia; out[o + 2] = qr; out[o + 3] = ib;
            out[o + 4] = mid; out[o + 5] = wd; out[o + 6] = ya; out[o + 7] = yb;   // (4,5) 는 아래에서 다듬은 뒤 x0,x1 로 바꾼다
            n++;
        }
        for (int it = 0; it < 4; it++) {
            for (int j = 1; j < n - 1; j++) {
                out[j * 8 + 4] = (out[(j - 1) * 8 + 4] + 2 * out[j * 8 + 4] + out[(j + 1) * 8 + 4]) / 4;
                out[j * 8 + 5] = (out[(j - 1) * 8 + 5] + 2 * out[j * 8 + 5] + out[(j + 1) * 8 + 5]) / 4;
            }
        }
        for (int j = 0; j < n; j++) {
            int o = j * 8;
            float mid = out[o + 4], wd = out[o + 5], ya = out[o + 6], yb = out[o + 7];
            out[o + 4] = mid - wd / 2; out[o + 5] = ya - 0.35f; out[o + 6] = mid + wd / 2; out[o + 7] = yb + 0.35f;
        }
        return n;
    }

package com.dullyyj.subway;

import static com.dullyyj.subway.SceneUtil.*;

/**
 * 길동무 패밀리 전용 — 친구들이 잠자는 시간에 보이는 집 안 배경 7곳(거실 · 이층침대 방 · 온돌방 · 다락방 · 텐트 방 · 별빛 창가방 · 책 읽는 방).
 * 어느 방을 보일지는 엔진(가족 Engine)이 look.place = "kroom0" ~ "kroom6" 으로 정한다. 여기는 받은 이름대로 벽 · 가구 · 바닥을 그릴 뿐이다.
 * 창밖 하늘은 시간대 색(L.c)을 그대로 쓰고, 날씨 효과(비 · 눈 · 별똥별 · 구름)는 그리지 않는다(엔진이 look 에서 날씨를 맑음으로 고쳐 보낸다).
 * 길동무 원본(SceneArt/ScenePlaces)은 고치지 않는다. android.* 를 쓰지 않는다.
 */
final class KidsRooms {
    private KidsRooms() {}

    static final int N = 7;
    static final String[] NAMES = {"거실", "이층침대 방", "온돌방", "다락방", "텐트 방", "별빛 창가방", "책 읽는 방"};

    static boolean is(SceneArt.Pal L) { return L != null && L.place != null && L.place.startsWith("kroom"); }
    static int idx(SceneArt.Pal L) {
        try { return Math.floorMod(Integer.parseInt(L.place.substring(5)), N); } catch (RuntimeException e) { return 0; }
    }

    // ── 밝기: 밤엔 어둡게(등불만 따뜻하게) ──
    static double dk(SceneArt.Pal L) { return L.night ? .42 : "dawn".equals(L.slot) ? .26 : "evening".equals(L.slot) ? .16 : .04; }
    static int d(double k, int col) { return mix(col, 0xFF0a0e2c, k); }
    static int d(double k, String col) { return d(k, css(col)); }
    static void glow(SceneCv c, double x, double y, double r, int cr, int cg, int cb, double al) { ScenePlaces.glowSpot(c, x, y, r, cr, cg, cb, al); }

    public static SceneArt.Bg build(SceneCv root, SceneArt.Pal L, SceneArt.Geom g, double dpr, String season, boolean dark) {
        double W = g.W, H = g.H, hy = g.hy; int tw = (int) Math.ceil(W * 1.7);
        SceneArt.Bg bg = new SceneArt.Bg(); bg.tw = tw;
        bg.sky = root.newLayer(W, H, dpr); bg.far = root.newLayer(tw, H, dpr); bg.mid = root.newLayer(tw, H, dpr); bg.gnd = root.newLayer(W, H + g.ext, dpr);
        int k = idx(L); double dm = dk(L);
        Seeded rng = new Seeded(k * 7919 + L.slot.length() * 131 + 17);
        SceneCv ws = bg.sky.cv(), fr = bg.far.cv(), md = bg.mid.cv(), fl = bg.gnd.cv();
        switch (k) {
            case 0: living(ws, fr, md, fl, L, g, dm, rng); break;
            case 1: bunk(ws, fr, md, fl, L, g, dm, rng); break;
            case 2: ondol(ws, fr, md, fl, L, g, dm, rng); break;
            case 3: attic(ws, fr, md, fl, L, g, dm, rng); break;
            case 4: tent(ws, fr, md, fl, L, g, dm, rng); break;
            case 5: starRoom(ws, fr, md, fl, L, g, dm, rng); break;
            default: study(ws, fr, md, fl, L, g, dm, rng);
        }
        return bg;
    }

    // ═══ 공통 도구 ═══
    static void wall(SceneCv c, SceneArt.Geom g, double dm, int top, int bot) {
        SceneCv.Grad gr = c.linear(0, 0, 0, g.hy + 4).stop(0, d(dm, top)).stop(1, d(dm, bot)); c.fillGrad(gr); c.fillRect(0, 0, g.W, g.H);
    }
    /** 벽 아래 걸레받이 + 바닥 경계(바닥 그림이 hy+3 부터 덮는다) */
    static void baseboard(SceneCv c, SceneArt.Geom g, double dm, String col) {
        c.fillStyle(d(dm, col)); c.fillRect(0, g.hy - 7, g.W, 8);
        c.fillStyle(rgba(0, 0, 0, .14)); c.fillRect(0, g.hy + 1, g.W, 3);
    }
    /** 바닥: 원근 가로줄 + 판자 이음 */
    static void floor(SceneCv c, SceneArt.Geom g, double dm, String near, String far, boolean planks, double seamAlpha) {
        double W = g.W, H = g.H, hy = g.hy, bot = H + g.ext;
        SceneCv.Grad gr = c.linear(0, hy, 0, bot).stop(0, d(dm, far)).stop(1, d(dm, near)); c.fillGrad(gr); c.fillRect(0, hy + 3, W, bot - hy - 3);
        c.strokeStyle(rgba(0, 0, 0, seamAlpha)); c.lineWidth(1);
        for (double yy : SceneArt.gridRows(hy + 3, H - hy - 3, 1.6, 6, H, g.ext)) { c.beginPath(); c.moveTo(0, yy); c.lineTo(W, yy); c.stroke(); }
        if (planks) for (int i = -10; i <= 10; i++) { c.beginPath(); c.moveTo(W / 2 + i * 22, hy + 3); c.lineTo(W / 2 + i * (22 + 40 * (bot - hy) / (H - hy)), bot); c.stroke(); }
        SceneCv.Grad sh = c.linear(0, hy + 3, 0, hy + 30).stop(0, rgba(0, 0, 0, .20)).stop(1, rgba(0, 0, 0, 0)); c.fillGrad(sh); c.fillRect(0, hy + 3, W, 27);
    }
    static void rug(SceneCv c, double cx, double cy, double rx, double ry, String a, String b, double dm) {
        c.fillStyle(rgba(0, 0, 0, .12)); c.beginPath(); c.ellipse(cx + 2, cy + 3, rx, ry, 0, 0, PI2); c.fill();
        c.fillStyle(d(dm, a)); c.beginPath(); c.ellipse(cx, cy, rx, ry, 0, 0, PI2); c.fill();
        c.fillStyle(d(dm, b)); c.beginPath(); c.ellipse(cx, cy, rx * .72, ry * .72, 0, 0, PI2); c.fill();
        c.fillStyle(d(dm, a)); c.beginPath(); c.ellipse(cx, cy, rx * .42, ry * .42, 0, 0, PI2); c.fill();
    }
    static void shadow(SceneCv c, double cx, double y, double rx, double ry) { c.fillStyle(rgba(0, 0, 0, .16)); c.beginPath(); c.ellipse(cx, y, rx, ry, 0, 0, PI2); c.fill(); }

    /** 창문: 프레임 + 하늘(시간대 색) + 달/별 + 십자살 */
    static void window(SceneCv c, SceneArt.Pal L, double x, double y, double w, double h, double dm, boolean arch, String frame, Seeded rng) {
        c.fillStyle(d(dm, frame));
        if (arch) { c.beginPath(); c.moveTo(x - 4, y + h + 4); c.lineTo(x - 4, y + w / 2); c.arc(x + w / 2, y + w / 2, w / 2 + 4, PI, PI2); c.lineTo(x + w + 4, y + h + 4); c.closePath(); c.fill(); }
        else { rr(c, x - 4, y - 4, w + 8, h + 8, 5); c.fill(); }
        c.save();
        if (arch) { c.beginPath(); c.moveTo(x, y + h); c.lineTo(x, y + w / 2); c.arc(x + w / 2, y + w / 2, w / 2, PI, PI2); c.lineTo(x + w, y + h); c.closePath(); } else rr(c, x, y, w, h, 3);
        c.clip();
        SceneCv.Grad gr = c.linear(0, y, 0, y + h).stop(0, d(dm * .35, L.c[0])).stop(.6, d(dm * .35, L.c[1])).stop(1, d(dm * .35, L.c[2])); c.fillGrad(gr); c.fillRect(x, y, w, h);
        if (L.night || "dawn".equals(L.slot)) {
            c.fillStyle("rgba(255,255,255,.85)");
            for (int i = 0; i < 14; i++) { double sx = x + rng.r() * w, sy = y + rng.r() * h * .8, sr = .6 + rng.r() * 1.1; c.beginPath(); c.arc(sx, sy, sr, 0, PI2); c.fill(); }
            if (L.moon) { c.fillStyle("#f4f1dc"); c.beginPath(); c.arc(x + w * .68, y + h * .24, Math.min(w, h) * .11, 0, PI2); c.fill(); }
        } else { c.fillStyle("rgba(255,255,255,.7)"); c.beginPath(); c.ellipse(x + w * .3, y + h * .3, w * .2, h * .06, 0, 0, PI2); c.ellipse(x + w * .42, y + h * .27, w * .13, h * .05, 0, 0, PI2); c.fill(); }
        c.restore();
        c.strokeStyle(d(dm, frame)); c.lineWidth(2.4);
        c.beginPath(); c.moveTo(x + w / 2, y); c.lineTo(x + w / 2, y + h); c.moveTo(x, y + h * .52); c.lineTo(x + w, y + h * .52); c.stroke();
        c.fillStyle(d(dm, shade(css(frame), -.12))); c.fillRect(x - 8, y + h + 3, w + 16, 4);
    }
    static void curtain(SceneCv c, double x, double y, double w, double h, String col, double dm, boolean left) {
        c.fillStyle(d(dm, col)); c.beginPath(); c.moveTo(x, y); c.lineTo(x + w, y); c.lineTo(left ? x + w * .78 : x + w, y + h); c.lineTo(left ? x : x + w * .22, y + h); c.closePath(); c.fill();
        c.strokeStyle(rgba(0, 0, 0, .12)); c.lineWidth(1);
        for (int i = 1; i < 4; i++) { c.beginPath(); c.moveTo(x + w * i / 4, y); c.lineTo(x + w * i / 4 + (left ? -2 : 2), y + h); c.stroke(); }
        c.fillStyle(d(dm, "#b8956a")); c.fillRect(x - 3, y - 4, w + 6, 4);
    }
    static void frameArt(SceneCv c, double x, double y, double w, double h, double dm, String frame, int a, int b) {
        c.fillStyle(d(dm, frame)); rr(c, x, y, w, h, 2); c.fill();
        c.fillStyle(d(dm, a)); c.fillRect(x + 3, y + 3, w - 6, h - 6);
        c.fillStyle(d(dm, b)); c.beginPath(); c.moveTo(x + 3, y + h - 3); c.lineTo(x + w * .38, y + h * .45); c.lineTo(x + w * .62, y + h * .7); c.lineTo(x + w * .78, y + h * .5); c.lineTo(x + w - 3, y + h - 3); c.closePath(); c.fill();
        c.fillStyle(d(dm, "#ffe9a8")); c.beginPath(); c.arc(x + w * .72, y + h * .3, Math.min(w, h) * .09, 0, PI2); c.fill();
    }
    static void bookshelf(SceneCv c, double x, double base, double w, double h, double dm, String wood, Seeded rng, int rows) {
        double top = base - h;
        c.fillStyle(rgba(0, 0, 0, .13)); c.fillRect(x - 2, top + 3, w + 6, h);
        c.fillStyle(d(dm, wood)); c.fillRect(x, top, w, h);
        c.fillStyle(d(dm, shade(css(wood), -.25))); c.fillRect(x + 3, top + 3, w - 6, h - 6);
        String[] bc = {"#d9534f", "#f0ad4e", "#5bc0de", "#5cb85c", "#9b6bd1", "#e86f9f", "#3f6fb5", "#e8d28a"};
        double rh = (h - 6) / rows;
        for (int r = 0; r < rows; r++) {
            double ry = top + 3 + r * rh; double bx = x + 5;
            while (bx < x + w - 9) {
                double bw = 3 + rng.r() * 5, bh = rh * (.55 + rng.r() * .38);
                c.fillStyle(d(dm, bc[rng.pick(bc.length)])); c.fillRect(bx, ry + rh - 3 - bh, bw, bh); bx += bw + .6;
            }
            c.fillStyle(d(dm, wood)); c.fillRect(x + 2, ry + rh - 3, w - 4, 3);
        }
    }
    static void sofa(SceneCv c, double x, double base, double w, double h, String col, double dm) {
        int a = d(dm, col), b = d(dm, shade(css(col), .12)), s = d(dm, shade(css(col), -.18));
        shadow(c, x + w / 2, base + 3, w * .55, 5);
        c.fillStyle(s); rr(c, x, base - h * .95, w, h * .8, 9); c.fill();
        c.fillStyle(a); rr(c, x + w * .08, base - h * .98, w * .84, h * .5, 8); c.fill();
        c.fillStyle(b); rr(c, x + w * .06, base - h * .52, w * .88, h * .36, 8); c.fill();
        c.fillStyle(s); rr(c, x - 2, base - h * .62, w * .12, h * .6, 6); c.fill(); rr(c, x + w * .9, base - h * .62, w * .12, h * .6, 6); c.fill();
        c.fillStyle(d(dm, "#5a4636")); c.fillRect(x + 4, base - h * .06, 4, h * .08); c.fillRect(x + w - 8, base - h * .06, 4, h * .08);
        c.strokeStyle(rgba(0, 0, 0, .12)); c.lineWidth(1); c.beginPath(); c.moveTo(x + w / 2, base - h * .5); c.lineTo(x + w / 2, base - h * .16); c.stroke();
    }
    static void cushion(SceneCv c, double x, double y, double s, String col, double dm, double rot) {
        c.save(); c.translate(x, y); c.rotate(rot); c.fillStyle(d(dm, col)); rr(c, -s / 2, -s / 2, s, s, s * .22); c.fill();
        c.fillStyle(rgba(255, 255, 255, .16)); rr(c, -s * .38, -s * .38, s * .5, s * .22, s * .1); c.fill(); c.restore();
    }
    static void lampFloor(SceneCv c, double x, double base, double h, double dm, double glowAlpha) {
        glow(c, x, base - h * .86, h * 1.05, 255, 214, 140, glowAlpha);
        c.fillStyle(d(dm * .5, "#6b4f35")); c.fillRect(x - 1.4, base - h * .82, 2.8, h * .82); c.fillRect(x - 7, base - 3, 14, 3.4);
        c.fillStyle(d(dm * .35, "#fff1c4")); c.beginPath(); c.moveTo(x - 10, base - h * .7); c.lineTo(x - 5, base - h); c.lineTo(x + 5, base - h); c.lineTo(x + 10, base - h * .7); c.closePath(); c.fill();
    }
    static void plant(SceneCv c, double x, double base, double h, double dm) {
        c.fillStyle(d(dm, "#c77b50")); c.beginPath(); c.moveTo(x - h * .16, base - h * .26); c.lineTo(x + h * .16, base - h * .26); c.lineTo(x + h * .11, base); c.lineTo(x - h * .11, base); c.closePath(); c.fill();
        String[] lc = {"#3f9a5a", "#4fa866", "#62b878"};
        for (int i = 0; i < 6; i++) { double a = -PI / 2 + (i - 2.5) * .34; c.fillStyle(d(dm, lc[i % 3])); c.beginPath(); c.ellipse(x + Math.cos(a) * h * .26, base - h * .3 + Math.sin(a) * h * .42, h * .08, h * .22, a + PI / 2, 0, PI2); c.fill(); }
    }
    static void clock(SceneCv c, double x, double y, double r, double dm) {
        c.fillStyle(d(dm, "#7a5a3a")); c.beginPath(); c.arc(x, y, r + 2.5, 0, PI2); c.fill();
        c.fillStyle(d(dm * .5, "#fff7e6")); c.beginPath(); c.arc(x, y, r, 0, PI2); c.fill();
        c.strokeStyle(d(dm, "#3a2e26")); c.lineWidth(1.6); c.beginPath(); c.moveTo(x, y); c.lineTo(x, y - r * .62); c.moveTo(x, y); c.lineTo(x + r * .42, y + r * .18); c.stroke();
    }
    static void stars(SceneCv c, double W, double hy, Seeded rng, int n, double dm) {
        c.fillStyle(rgba(255, 238, 170, .75 - dm * .3));
        for (int i = 0; i < n; i++) { double sx = rng.r() * W, sy = 8 + rng.r() * (hy - 40); starShape(c, sx, sy, 2.2 + rng.r() * 2.2); }
    }
    static void starShape(SceneCv c, double x, double y, double r) {
        c.beginPath(); for (int i = 0; i < 10; i++) { double a = -PI / 2 + i * PI / 5, rr2 = i % 2 == 0 ? r : r * .45; double px = x + Math.cos(a) * rr2, py = y + Math.sin(a) * rr2; if (i == 0) c.moveTo(px, py); else c.lineTo(px, py); } c.closePath(); c.fill();
    }
    static void fairy(SceneCv c, double x0, double x1, double y, double sag, double dm, Seeded rng, int n) {
        c.strokeStyle(d(dm, "#5a4636")); c.lineWidth(1); c.beginPath(); c.moveTo(x0, y);
        for (int i = 1; i <= 24; i++) { double u = i / 24.0; c.lineTo(x0 + (x1 - x0) * u, y + Math.sin(u * PI) * sag); } c.stroke();
        String[] cc = {"#ffd86b", "#ff9fb8", "#9fd8ff", "#b6f2a4"};
        for (int i = 0; i < n; i++) { double u = (i + .5) / n, bx = x0 + (x1 - x0) * u, by = y + Math.sin(u * PI) * sag + 4; String col = cc[i % 4]; int rgb = css(col);
            glow(c, bx, by, 15, (rgb >> 16) & 255, (rgb >> 8) & 255, rgb & 255, .55); c.fillStyle(css(col)); c.beginPath(); c.arc(bx, by, 2.4, 0, PI2); c.fill(); }
    }
    static void toyBlocks(SceneCv c, double x, double base, double s, double dm) {
        String[] cc = {"#e85d5d", "#4d9de0", "#f2c14e", "#5fbf7a"};
        for (int i = 0; i < 3; i++) { c.fillStyle(d(dm, cc[i])); c.fillRect(x + i * (s + 1), base - s, s, s); c.fillStyle(rgba(255, 255, 255, .22)); c.fillRect(x + i * (s + 1), base - s, s, 2); }
        c.fillStyle(d(dm, cc[3])); c.fillRect(x + (s + 1) * .5, base - s * 2 - 1, s, s);
    }

    // ═══ 0 거실 ═══
    static void living(SceneCv w, SceneCv far, SceneCv mid, SceneCv fl, SceneArt.Pal L, SceneArt.Geom g, double dm, Seeded rng) {
        double W = g.W, hy = g.hy; double u = Math.max(.7, hy / 190);
        wall(w, g, dm, css("#efdcc0"), css("#e3c9a4")); baseboard(w, g, dm, "#fbf3e4");
        w.fillStyle(rgba(120, 80, 40, .06)); for (double x = 0; x < W; x += 26) w.fillRect(x, 0, 13, hy);
        window(w, L, W * .64, hy * .2, W * .24, hy * .5, dm, false, "#fbf3e4", rng);
        frameArt(w, W * .12, hy * .22, W * .13, hy * .26, dm, "#8a6a4a", css("#a9d4f0"), css("#5fa36a"));
        frameArt(w, W * .29, hy * .3, W * .1, hy * .2, dm, "#8a6a4a", css("#f6c9a6"), css("#c97a5a"));
        clock(w, W * .5, hy * .2, 9 * u, dm);
        floor(fl, g, dm, "#c8935f", "#a97a4c", true, .1);
        rug(fl, W * .45, hy + (g.H - hy) * .44, W * .3, (g.H - hy) * .2, "#cf6b6b", "#f1d0a0", dm);
        curtain(mid, W * .59, hy * .12, W * .06, hy * .66, "#d98a8a", dm, true); curtain(mid, W * .88, hy * .12, W * .06, hy * .66, "#d98a8a", dm, false);
        bookshelf(far, W * .02, hy + 4, W * .13, hy * .78, dm, "#8a6a4a", rng, 4);
        sofa(mid, W * .19, hy + 4, W * .31, hy * .42, "#6f8fb5", dm);
        cushion(mid, W * .27, hy - hy * .26, 17 * u, "#f2c14e", dm, -.18); cushion(mid, W * .42, hy - hy * .26, 17 * u, "#e07a7a", dm, .16);
        plant(mid, W * .965, hy + 4, hy * .46, dm);
        lampFloor(mid, W * .54, hy + 5, hy * .78, dm, .8);
        glow(fl, W * .54, hy + 30, hy * .6, 255, 214, 140, .22 * (1 - dm * .6));
    }

    // ═══ 1 이층침대 방 ═══
    static void bunk(SceneCv w, SceneCv far, SceneCv mid, SceneCv fl, SceneArt.Pal L, SceneArt.Geom g, double dm, Seeded rng) {
        double W = g.W, hy = g.hy; double u = Math.max(.7, hy / 190);
        wall(w, g, dm, css("#bcd9ef"), css("#a9c9e6")); baseboard(w, g, dm, "#ffffff");
        w.fillStyle(rgba(255, 255, 255, .5)); stars(w, W, hy, rng, 16, dm);
        window(w, L, W * .7, hy * .16, W * .22, hy * .42, dm, false, "#ffffff", rng);
        frameArt(w, W * .46, hy * .2, W * .1, hy * .2, dm, "#ffffff", css("#ffe9a8"), css("#6fbf73"));
        floor(fl, g, dm, "#d9b88c", "#c39f72", true, .08);
        rug(fl, W * .55, hy + (g.H - hy) * .5, W * .28, (g.H - hy) * .19, "#7fb2e0", "#fff0b8", dm);
        // 이층침대
        double bx = W * .03, bw = W * .4, bh = hy * .92, base = hy + 4;
        shadow(mid, bx + bw / 2, base + 3, bw * .55, 5);
        mid.fillStyle(d(dm, "#c28a55")); mid.fillRect(bx, base - bh, 6, bh); mid.fillRect(bx + bw - 6, base - bh, 6, bh);
        for (int lv = 0; lv < 2; lv++) {
            double ly = base - bh * (.26 + lv * .46);
            mid.fillStyle(d(dm, "#c28a55")); mid.fillRect(bx, ly, bw, 5);
            mid.fillStyle(d(dm, lv == 0 ? "#f4f0ff" : "#fff6e0")); rr(mid, bx + 7, ly - bh * .1, bw - 14, bh * .12, 6); mid.fill();
            mid.fillStyle(d(dm, lv == 0 ? "#7d6bd1" : "#e07a7a")); rr(mid, bx + bw * .28, ly - bh * .1, bw * .68, bh * .1, 6); mid.fill();
            mid.fillStyle(d(dm, "#fffdf6")); rr(mid, bx + 10, ly - bh * .13, bw * .2, bh * .09, 5); mid.fill();
        }
        mid.fillStyle(d(dm, "#c28a55")); mid.fillRect(bx, base - bh, bw, 5);
        mid.fillStyle(d(dm, "#a67140")); mid.fillRect(bx + bw - 5, base - bh * .7, 3, bh * .45);
        glow(mid, bx + bw * .5, base - bh * .78, bh * .55, 255, 220, 150, .4);
        bookshelf(far, W * .5, hy + 4, W * .17, hy * .52, dm, "#c28a55", rng, 3);
        toyBlocks(fl, W * .76, hy + (g.H - hy) * .12, 10 * u, dm);
        plant(mid, W * .95, hy + 4, hy * .5, dm);
    }

    // ═══ 2 온돌방 ═══
    static void ondol(SceneCv w, SceneCv far, SceneCv mid, SceneCv fl, SceneArt.Pal L, SceneArt.Geom g, double dm, Seeded rng) {
        double W = g.W, hy = g.hy; double u = Math.max(.7, hy / 190);

package com.dullyyj.subway;

import static com.dullyyj.subway.SceneUtil.*;
import static com.dullyyj.subway.SceneCore.po;

import com.dullyyj.subway.SceneCore.Actor;

/**
 * 길동무 키즈 전용 동작 — 길동무 장면 그리개(SceneCore)에 빌드할 때 끼워 넣는다(길동무 원본은 고치지 않는다).
 * 어떤 친구가 언제 무엇을 하는지는 엔진이 동작 '이름'(traits.acts)으로 정해 준다. 여기는 그 이름대로 몸짓과 말풍선을 그리는 곳뿐이다.
 * 자세 시트(<id>_pose): 0 앉기 · 1 잠 · 2 웃음 · 3 덜덜 · 5 찡그림.
 */
final class KidsActs {
    /** 장면 그림 배율(1 이면 길동무 그대로, 0.5 면 친구도 배경도 절반 크기로 더 넓게 보인다). */
    static final double ZOOM = 0.5;

    private KidsActs() {}

    /** 친구를 건드렸다. 공부 중이면 바로 화를 낸다. */
    static void poke(SceneCore S, Actor a) {
        if (a == null || a.held || a.dead) return;
        int pk = com.dullyyj.kids.Poke.kind(a.id);
        if (pk == 0) return;
        a.pr = null; a.cut = Double.NaN; a.sx = 1; a.sy = 1; a.ox = 0; a.ang = 0; a.stk = null; a.feetScale = false; a.heart = 0; a.chain = null;
        S.startAct(a, pk == 2 ? "kwake" : "kangry");
    }

    static void register(final SceneCore S) {
        // 공부: 앉아서 가만히 집중
        SceneActs.A_(S, "kstudy", 6, null, (a, u, dt, A) -> {
            double T = S.t; S.standF(a); boolean on = u > .06 && u < .94;
            a.pose = on ? 0 : -1; a.ang = on ? Math.sin(T * 1.6) * .015 : 0; a.sy = 1 + (on ? Math.sin(T * 2.2) * .01 : 0);
            if (u > .08 && S.once(A, "sy")) S.say(a, "공부 중이에요 📖", 4.8);
        }, null);
        // 잠자기: 엎드려 쿨쿨
        SceneActs.A_(S, "ksleep", 6, null, (a, u, dt, A) -> {
            double T = S.t; S.standF(a); a.pose = 1; a.sy = 1 + Math.sin(T * 1.7) * .014;
            if (S.every(A, "zz", .8)) S.ptx("zzz", a.x + a.dir * S.HH(a) * .3, S.feetY(a) - S.HH(a) * .8, 1, po().a(-PI * .6, -PI * .4).s(6, 9).g(-4).l(1, 1.4).r(1));
        }, null);
        // 같이 놀기: 폴짝폴짝 신나게
        SceneActs.A_(S, "kplay", 3.6, null, (a, u, dt, A) -> {
            double h = S.HH(a), T = S.t; S.standF(a); boolean on = u > .06 && u < .94;
            a.pose = on ? 2 : -1; a.y = on ? -Math.abs(Math.sin(T * 6.6)) * 8 : 0; a.sy = 1 + (on ? .03 * Math.sin(T * 13.2) : 0);
            if (S.every(A, "sp", .25)) S.ptx("spark", a.x + (S.rnd() - .5) * h * .9, S.feetY(a) - h * (.5 + S.rnd() * .6), 1, po().a(-PI * .8, -PI * .2).s(8, 16).g(0).l(.3, .5).r(.9));
            if (u > .05 && S.once(A, "sy")) S.say(a, new String[] {"같이 놀자!", "우리 같이 놀아요!", "신난다!", "야호!"}[(int) (S.rnd() * 4) % 4], 1.8);
        }, null);
        // 밥 먹기
        SceneActs.A_(S, "keat", 4, null, (a, u, dt, A) -> {
            double h = S.HH(a), T = S.t; S.standF(a); boolean on = u > .08 && u < .9;
            a.pose = on ? 0 : -1; a.y = on ? -Math.abs(Math.sin(T * 7)) * 2 : 0; a.sy = 1 + (on ? .02 * Math.sin(T * 14) : 0);
            if (u > .06 && S.once(A, "sy")) S.say(a, "냠냠! 맛있다!", 3.0);
            if (u > .9 && S.once(A, "hh")) S.ptx("heart", a.x, S.feetY(a) - h * .8, 3, po().a(-PI * .98, -PI * .02).s(24, 54).g(55).l(.9, 1.4).r(1.2).jx(5));
        }, null);
        // 배고파요
        SceneActs.A_(S, "kbeg", 4, null, (a, u, dt, A) -> {
            double T = S.t; S.standF(a); boolean on = u > .08 && u < .92;
            a.pose = on ? 0 : -1; a.y = on ? -Math.abs(Math.sin(T * 5)) * 3 : 0;
            if (u > .06 && S.once(A, "sy")) S.say(a, "배고파요! 밥 주세요", 3.4);
        }, null);
        // 삐졌어요
        SceneActs.A_(S, "ksulk", 4, null, (a, u, dt, A) -> {
            S.standF(a); boolean on = u > .08 && u < .92;
            a.pose = on ? 5 : -1;
            if (u > .06 && S.once(A, "sy")) S.say(a, "흥! 삐졌어요", 3.4);
        }, null);
        // 아파요
        SceneActs.A_(S, "ksick", 4, null, (a, u, dt, A) -> {
            double T = S.t; S.standF(a); boolean on = u > .08 && u < .92;
            a.pose = on ? 3 : -1; a.ox = on ? Math.sin(T * 30) * .8 : 0;
            if (u > .06 && S.once(A, "sy")) S.say(a, "아파요… 도와주세요", 3.4);
        }, null);
        // 응가했어요
        SceneActs.A_(S, "kpoop", 3.4, null, (a, u, dt, A) -> {
            S.standF(a); boolean on = u > .08 && u < .92;
            a.pose = on ? 0 : -1;
            if (u > .06 && S.once(A, "sy")) S.say(a, "응가했어요! 💩", 2.8);
        }, null);
        // 공부 중에 건드리면 화낸다
        SceneActs.A_(S, "kangry", 4.2, null, (a, u, dt, A) -> {
            double h = S.HH(a), T = S.t, tt = u * 4.2, amp = 1 - ease(clamp((tt - 3.5) / .7, 0, 1)); S.standF(a); a.pose = 5; a.cell = amp < .3 ? 14 : (int) (T * 7) % 2 == 0 ? 15 : 9;
            double b = Math.abs(Math.sin(T * 9.5)); a.y = -b * h * .16 * amp; a.sy = 1 - .12 * (1 - b) * amp; a.sx = 1 + .08 * (1 - b) * amp; a.ox = Math.sin(T * 38) * 1.4 * amp; a.ang = Math.sin(T * 19) * .05 * amp;
            if (S.once(A, "st")) { S.stk(a, "angry", 3.4); S.haptic("angry"); }
            if (S.every(A, "say", 1.3)) S.say(a, new String[] {"공부 중이잖아!!", "건드리지 마!", "집중해야 해!", "시끄러워!"}[(int) ((S.t * 3) % 4)], 1.3);
            if (S.every(A, "pf", .17)) S.ptx("puff", a.x + (S.rnd() - .5) * h * .5, S.feetY(a) - h * 1.0, 1, po().a(-PI * .85, -PI * .15).s(14, 30).g(-10).l(.4, .7).r(1.1).jx(3));
        }, null);
        // 자는 중에 깨우면 화내고, 다시 잔다
        SceneActs.A_(S, "kwake", 4.2, null, (a, u, dt, A) -> {
            double h = S.HH(a), T = S.t, tt = u * 4.2, amp = 1 - ease(clamp((tt - 3.5) / .7, 0, 1)); S.standF(a); a.pose = 5; a.cell = amp < .3 ? 14 : (int) (T * 7) % 2 == 0 ? 15 : 9;
            double b = Math.abs(Math.sin(T * 9.5)); a.y = -b * h * .16 * amp; a.sy = 1 - .12 * (1 - b) * amp; a.sx = 1 + .08 * (1 - b) * amp; a.ox = Math.sin(T * 38) * 1.4 * amp; a.ang = Math.sin(T * 19) * .05 * amp;
            if (S.once(A, "st")) { S.stk(a, "angry", 3.4); S.haptic("angry"); }
            if (S.every(A, "say", 1.3)) S.say(a, new String[] {"깨우지 마… 졸려!", "더 잘 거야!", "으… 왜 깨워!", "쿨쿨 중이었어!"}[(int) ((S.t * 3) % 4)], 1.3);
            if (S.every(A, "pf", .17)) S.ptx("puff", a.x + (S.rnd() - .5) * h * .5, S.feetY(a) - h * 1.0, 1, po().a(-PI * .85, -PI * .15).s(14, 30).g(-10).l(.4, .7).r(1.1).jx(3));
        }, null);
    }
}

package com.dullyyj.subway;

import android.graphics.Bitmap;

import com.dullyyj.kids.Art;
import com.dullyyj.kids.Dress;
import com.dullyyj.kids.WearImg;
import com.dullyyj.kids.Wardrobe;

/**
 * 길동무 키즈 꾸미기 — 홈 장면에서 돌아다니는 친구들이 꾸민 모습 그대로 다닌다.
 * 무엇을 입었는지(Dress)와 몸의 어디에 얼마 크게 놓을지(Wardrobe.place)는 엔진이 계산해 두고, 여기는 장면 도화지(SceneCv)에 붓질만 한다.
 * 친구를 그리는 같은 변환(좌우 뒤집기 · 기울기 · 눌림) 안에서 그리므로 몸을 따라 함께 움직인다.
 */
final class KidsWear {
    private KidsWear() {}

    private static final class Pen implements Art.Pen {
        SceneCv c;
        @Override public void begin() { c.beginPath(); }
        @Override public void move(float x, float y) { c.moveTo(x, y); }
        @Override public void line(float x, float y) { c.lineTo(x, y); }
        @Override public void quad(float cx, float cy, float x, float y) { c.quadTo(cx, cy, x, y); }
        @Override public void close() { c.closePath(); }
        @Override public void oval(float cx, float cy, float rx, float ry) { c.ellipse(cx, cy, rx, ry, 0, 0, Math.PI * 2); }
        @Override public void fill(int argb) { c.fillStyle(argb); c.fill(); }
        @Override public void stroke(int argb, float w) { c.strokeStyle(argb); c.lineWidth(w); c.stroke(); }
    }

    private static final Pen PEN = new Pen();
    private static final float[] O = new float[4], SP = new float[3], ST = new float[8 * Wardrobe.NS];
    private static final Bitmap[] BM = new Bitmap[8];
    private static final AndroidCv.AImg[] IM = new AndroidCv.AImg[8];
    private static final Bitmap[] WBM = new Bitmap[8 * 2 * Wardrobe.PER];
    private static final AndroidCv.AImg[] WIM = new AndroidCv.AImg[8 * 2 * Wardrobe.PER];
    private static final int[] WT = new int[6];

    /** 몸에 입혀 구운 옷 한 벌의 시트. 없거나 아직 읽는 중이면 null. */
    private static AndroidCv.AImg worn(int ci, int i) {
        Bitmap b = WearImg.worn(ci, i);
        if (b == null) return null;
        int k = ci * 2 * Wardrobe.PER + i;
        if (WBM[k] != b) { WBM[k] = b; WIM[k] = new AndroidCv.AImg(b); }
        return WIM[k];
    }

    /** 3D 로 그려 둔 입을 것 시트(없거나 아직 읽는 중이면 null → 그림 문장으로 그린다). */
    private static AndroidCv.AImg img(int ci) {
        Bitmap b = WearImg.atlas(ci);
        if (b == null) return null;
        if (BM[ci] != b) { BM[ci] = b; IM[ci] = new AndroidCv.AImg(b); }
        return IM[ci];
    }

    /** kind: 0 걷기 · 1 자세 · 2 동작 시트. (dx, dy) = 칸 왼쪽 위(변환된 자리), sc = 칸 한 픽셀의 장면 크기. */
    static void draw(SceneCv c, SceneCore.Actor a, int kind, int idx, double dx, double dy, double sc) {
        if (a == null || a.intro || a.heart > 0 || a.id == null) return;
        Dress.Look lk = Dress.I.look(a.id);
        if (lk == null) return;
        int ci = Wardrobe.ci(a.id);
        if (ci < 0) return;
        PEN.c = c;
        c.lineCapRound(true);
        AndroidCv.AImg at = img(ci);
        for (int n = 0; n < 3; n++) {
            int slot = n == 0 ? 1 : n == 1 ? 2 : 0;          // 옷 → 소지품 → 모자
            Art.Op[] ops = lk.ops[slot];
            if (ops == null || !(slot == 2 ? Wardrobe.floatAt(ci, kind, idx, android.os.SystemClock.uptimeMillis(), O) : Wardrobe.place(ci, kind, idx, slot, O))) continue;   // 소지품은 친구 옆에서 둥실둥실(엔진 floatAt)
            int i = lk.idx[slot];
            if (at != null && Wardrobe.sprite(ci, slot, i, SP)) {
                // 몸에 입혀 구운 옷(칸마다 한 장): 읽혀 있으면 그것만 그린다
                int bi = slot == 0 ? Wardrobe.PER + i : i;         // 구운 모자는 번호 20 + 모자 번호
                AndroidCv.AImg wi = Wardrobe.baked(ci, slot, i) ? worn(ci, bi) : null;
                if (wi != null && Wardrobe.wornTile(ci, kind, idx, bi, WT)) {
                    c.drawImage(wi, WT[0], WT[1], WT[2], WT[3], dx + WT[4] * sc, dy + WT[5] * sc, WT[2] * sc, WT[3] * sc);
                    continue;
                }
                // 옷은 몸 윤곽을 따라 줄마다 몸 폭에 맞춰 입힌다(엔진이 줄 · 자리를 계산) — 나머지는 한 장
                int nn = Wardrobe.strips(ci, kind, idx, slot, i, O[0], O[1], O[2], ST);
                int q = slot * Wardrobe.PER + i, cs = Wardrobe.CELL, ox = (q % Wardrobe.GRID) * cs, oy = (q / Wardrobe.GRID) * cs;
                // 모자는 머리 기울기를 따라 살짝 기울인다(엔진이 각을 계산, 여기는 돌려 그리기만)
                float tilt = slot == 0 ? Wardrobe.hatTilt(ci, kind, idx) : slot == 2 ? O[3] : 0f;
                boolean rot = tilt != 0f && nn > 0;
                if (rot) {
                    double px = dx + (ST[4] + ST[6]) * 0.5 * sc, py = dy + (ST[5] + (ST[7] - ST[5]) * (slot == 2 ? 0.5 : 0.8)) * sc;
                    c.save(); c.translate(px, py); c.rotate(tilt); c.translate(-px, -py);
                }
                for (int j = 0; j < nn; j++) {
                    int o = j * 8;
                    c.drawImage(at, ox + (int) ST[o], oy + (int) ST[o + 1], (int) ST[o + 2] - (int) ST[o], (int) ST[o + 3] - (int) ST[o + 1],
                        dx + ST[o + 4] * sc, dy + ST[o + 5] * sc, (ST[o + 6] - ST[o + 4]) * sc, (ST[o + 7] - ST[o + 5]) * sc);
                }
                if (rot) c.restore();
            } else {
                Art.draw(PEN, ops, (float) (dx + O[0] * sc), (float) (dy + O[1] * sc), (float) (O[2] * sc));
            }
        }
    }
}

    // ───────── 친구가 늘 화면에 머문다 ─────────
    /** 한 번에 최대 4마리만 나온다. 4마리가 모두 밥·응가 때문에 멈춰 있으면 모두 지나가고 다른 친구들로 바뀐다(rotate). */
    static final int MAX_ON = 4;
    static int want(SceneCore S) {
        int n = S.res != null && S.res.cast != null ? S.res.cast.size() : 3;
        return Math.max(1, Math.min(MAX_ON, n));
    }
    static int slotOf(SceneCore S, Actor o, double lo, double span) {
        return (int) clamp(Math.round((o.x - lo) / span * MAX_ON - .5), 0, MAX_ON - 1);
    }
    /** 4칸 중 비어 있는 곳에 선다. 새로 들어오는 중이면(rotate 직후) 화면 가장자리에서 걸어 들어온다. */
    static double startX(SceneCore S, Actor a, double edgeX) {
        double lo = a.w * .7, span = Math.max(10, S.W - a.w * 1.4);
        boolean[] used = new boolean[MAX_ON];
        for (Actor o : S.actors) if (o != a && !o.hero && !o.leaving && !o.dead) used[slotOf(S, o, lo, span)] = true;
        int ixa = ix(a.id), slot = -1;
        for (int k = 0; k < MAX_ON && slot < 0; k++) { int c = (ixa + k * 2 + k / 2) % MAX_ON; if (!used[c]) slot = c; }
        if (slot < 0) slot = ixa % MAX_ON;
        double x = lo + span * (slot + .5) / MAX_ON + (S.R.next() - .5) * 10;
        if (S.t < enterEdgeUntil) return edgeX;
        return x;
    }
    static double enterEdgeUntil = 0;
    static final HashMap<String, Double> rest = new HashMap<>();        // 방금 지나간 친구는 한동안 쉰다(다른 친구가 먼저 나온다)
    static final HashMap<String, Double> blockedSince = new HashMap<>();
    static boolean blocked(String id) {
        String s = Poke.state(id);
        return s.equals("beg") || s.equals("poop");
    }
    static double restLeft(SceneCore S, String id) {
        Double r = rest.get(id);
        if (r == null) return 0;
        double l = r - S.t;
        return l > 0 && l <= 41.0 ? l : 0;
    }
    static boolean visible(SceneCore S, String id) {
        for (Actor a : S.actors) if (id.equals(a.id) && !a.leaving && !a.dead) return true;
        return false;
    }
    /** 다시 부를 수 있는 친구인가(spawn 후보 거르기) — 방금 지나간 친구는 쉬는 동안 뒤로 미루고(모두 쉬면 가장 오래 쉰 친구부터),
     *  멈춰 있는 친구보다 움직일 수 있는 친구가 있으면 그 친구를 먼저 부른다. */
    static boolean avail(SceneCore S, String cand) {
        if (S.res == null || S.res.cast == null) return true;
        double rc = restLeft(S, cand);
        boolean bc = blocked(cand);
        for (String o : S.res.cast) {
            if (o.equals(cand) || visible(S, o)) continue;
            double ro = restLeft(S, o);
            if (ro < rc - .001) return false;
            if (ro <= rc + .001 && bc && !blocked(o)) return false;
        }
        return true;
    }
    /** 보이는 친구 전부가 밥·응가로 멈춰 있으면(잠깐 지켜본 뒤) 모두 지나가게 하고 다른 친구들을 부른다. */
    static void rotate(SceneCore S) {
        int cast = S.res != null && S.res.cast != null ? S.res.cast.size() : 0;
        if (cast <= MAX_ON || menuA != null) return;
        int on = 0; double latest = -1;
        for (Actor a : S.actors) {
            if (a.hero || a.dead || a.leaving || a.id == null) continue;
            on++;
            Double bs = blockedSince.get(a.id);
            if (bs == null || !blocked(a.id)) return;
            latest = Math.max(latest, bs);
        }
        if (on < MAX_ON || S.t - latest < 8.0) return;
        for (Actor a : S.actors) {
            if (a.hero || a.dead || a.leaving || a.id == null) continue;
            reset(a); a.act = null; a.cell = -1; a.bub = ""; a.sk = null;
            a.leaving = true; a.st = "walk"; a.tx = a.x < S.W / 2 ? -a.w * 1.1 : S.W + a.w * 1.1; a.tz = a.z; a.onArrive = null;
            rest.put(a.id, S.t + 40.0);
            blockedSince.remove(a.id);
        }
        enterEdgeUntil = S.t + 3.0;
        S.spawnAt = S.t + .4;
    }
    /** 말풍선이 한꺼번에 뜨지 않게 친구마다 첫 말을 조금씩 어긋나게 한다. */
    static void stagger(SceneCore S, SceneCore.AS A, Actor a, String key, double gap) {
        if (S.once(A, "stg")) A.ev.put(key, S.t - gap + ix(a.id) * gap / 8.0);
    }

        S.startAct(a, an);
        return true;
    }

        if (au != null && S.t < au) return;                       // 방금 건드려서 화난 친구에게는 메뉴가 뜨지 않는다

    static void closeMenu() { menuA = null; }
    static double[] menuCenter(SceneCore S) {
        Actor a = menuA;
        double cx = a.x, cy = S.feetY(a) - S.HH(a) * .5;
        double m = M_R + M_IR + 4;
        cx = clamp(cx, m, Math.max(m, S.W - m));
        cy = clamp(cy, m + 28, Math.max(m + 28, S.H - m));
        return new double[] {cx, cy};
    }
    static final double[] M_ANG = {-3 * PI / 4, -PI / 4, PI / 4, 3 * PI / 4};
    static boolean menuDown(SceneCore S, double x, double y) {
        if (menuA == null) return false;
        if (S.t - menuT0 < .25) return true;
        double[] c = menuCenter(S);
        for (int i = 0; i < 4; i++) {
            double ix = c[0] + Math.cos(M_ANG[i]) * M_R, iy = c[1] + Math.sin(M_ANG[i]) * M_R;
            if (Math.hypot(x - ix, y - iy) <= M_IR + 7) {
                String id = menuA.id;
                closeMenu();
                S.haptic("release");
                Poke.pick(id, M_ACT[i]);
                return true;
            }
        }
        closeMenu();
        return false;
    }

    // ───────── 매 프레임: 엔진이 정한 상태대로 동작을 맞춘다 ─────────
    static String shownLast = "";
    /** 화면 안에 나와 있는 친구들을 엔진에 알린다(바뀔 때만). */
    static void publishShown(SceneCore S) {
        java.util.TreeSet<String> ids = new java.util.TreeSet<String>();
        for (int i = 0; i < S.actors.size(); i++) {
            Actor a = S.actors.get(i);
            if (a.hero || a.dead || a.leaving || a.id == null) continue;
            if (a.x < 0 || a.x > S.W) continue;
            ids.add(a.id);
        }
        StringBuilder b = new StringBuilder(",");
        for (String id : ids) b.append(id).append(',');
        String s = b.toString();
        if (!s.equals(shownLast)) {
            shownLast = s;
            Poke.setShown(s);
        }
    }

    static void tick(SceneCore S, double dt) {
        publishShown(S);
        if (!S.running || S.tv >= 0 || S.pan >= 0 || S.hello != null || !"idle".equals(S.mode)) return;
        if (menuA != null && (S.t - menuT0 > MENU_TTL || !S.actors.contains(menuA) || menuA.held)) closeMenu();
        for (int i = 0; i < S.actors.size(); i++) {
            Actor a = S.actors.get(i);
            if (a.hero || a.dead || a.leaving || a.hidden || a.held || a.id == null) continue;
            String stt = a.st;
            if (stt.equals("fall") || stt.equals("held") || stt.equals("emerge") || stt.equals("pan") || stt.equals("hidden")) continue;
            if (a.x < a.w * .3 || a.x > S.W - a.w * .3) continue;       // 아직 화면 밖에서 걸어 들어오는 중이면 도착한 뒤에 동작을 시작한다
            String s = Poke.state(a.id);
            int n = Poke.poops(a.id);
            // 응가 개수 변화
            Integer pn = poopN.get(a.id);
            int pv = pn == null ? 0 : pn;
            if (pn == null && n > 0 && !poopT.containsKey(a.id)) {      // 장면이 새로 만들어졌는데 이미 응가가 있다 → 바로 보이게
                poopT.put(a.id, S.t - 10);
                poopSide.put(a.id, -a.dir);
            }
            if (pn != null && n > pv && poopT.containsKey(a.id)) {
                popT.put(a.id, S.t);
                S.ptx("puff", a.x - side(a) * S.HH(a) * .5, S.feetY(a) - 2, 4, po().a(-PI * .9, -PI * .1).s(10, 22).g(-6).l(.5, .9).jx(5));
            }
            if (n == 0 && poopT.containsKey(a.id)) {
                S.ptx("spark", a.x - side(a) * S.HH(a) * .5, S.feetY(a) - 6, 6, po().a(-PI * .9, -PI * .1).s(14, 30).g(20).l(.5, .9).r(1).jx(8));
                poopT.remove(a.id); popT.remove(a.id);
                if (a.act != null && a.act.n.equals("kpoop")) S.say(a, "깨끗해졌어요! 고마워요 ✨", 2.6);
            }
            poopN.put(a.id, n);
            // 화내는 중이면 그 동작을 이어 간다
            Double au = angryUntil.get(a.id);
            boolean angry = au != null && S.t < au;
            String name = angry ? angryAct.get(a.id) : STICKY.get(s);
            String cur = a.st.equals("act") && a.act != null ? a.act.n : "";
            if (S.bigAct(a) && !isKids(cur)) continue;
            if (name != null) {
                if (!name.equals(cur)) {
                    reset(a);
                    S.startAct(a, name);
                }
            } else {
                if (isKids(cur) && !cur.equals("kplay")) S.endAct(a);
                String last = applied.get(a.id);
                if ("play".equals(s) && !"play".equals(last) && !isKids(cur)) {
                    reset(a);
                    S.startAct(a, "kplay");
                }
            }
            applied.put(a.id, angry ? "angry" : s);
            boolean stuck = !angry && blocked(a.id) && (cur.equals("kbeg") || cur.equals("kpoop"));
            if (stuck) { if (!blockedSince.containsKey(a.id)) blockedSince.put(a.id, S.t); }
            else blockedSince.remove(a.id);
        }
        rotate(S);
    }
    static int side(Actor a) {
        Integer v = poopSide.get(a.id);
        return v == null ? -a.dir : v;
    }

    // ───────── 그리기: 친구 뒤(응가) · 앞(밥그릇·책) ─────────
    static void under(SceneCore S, SceneCv c, Actor a) {
        if (a == null || a.id == null || !a.vis || a.leaving) return;
        int n = Poke.poops(a.id);
        if (n <= 0 || !poopT.containsKey(a.id)) return;
        double h = S.HH(a), zz = SceneCore.zs(a.z), sc = h / 44.0, y = S.feetY(a), T = S.t;
        for (int i = 0; i < Math.min(3, n); i++) {
            double ox = side(a) * h * (.48 + i * .34), oy = (i == 1 ? 2.0 : i == 2 ? -1.5 : 0) * zz;
            double pt = popT.containsKey(a.id) && i == n - 1 ? S.t - popT.get(a.id) : 9;
            double since = i == 0 ? S.t - poopT.get(a.id) : 9;
            double q = Math.min(pt, since);
            double pop = q < .55 ? 1 + .35 * Math.sin(q / .55 * PI) * (1 - q / .55) - (q < .12 ? (1 - q / .12) * .8 : 0) : 1;
            drawPoop(c, a.x + ox, y + oy, sc * .95, Math.max(.05, pop), T + i);
        }
    }
    static void over(SceneCore S, SceneCv c, Actor a) {
        if (a == null || a.id == null || !a.vis || a.held || a.leaving) return;
        String s = Poke.state(a.id);
        double h = S.HH(a), sc = h / 44.0, y = S.feetY(a), T = S.t;
        boolean actNow = a.st.equals("act") && a.act != null;
        String cur = actNow ? a.act.n : "";
        if ("beg".equals(s) && cur.equals("kbeg")) {
            double bob = Math.sin(T * 3.2) * .6;
            drawBowl(c, a.x + a.dir * h * .62, y + 2 + bob, sc * .95, false, T);
        } else if (("eat".equals(s)) && cur.equals("keat")) {
            drawBowl(c, a.x + a.dir * h * .62, y + 2, sc * .95, true, T);
        } else if (("study".equals(s) || "angry".equals(s)) && cur.equals("kstudy")) {
            drawBook(c, a.x + a.dir * h * .5, y - h * .02, sc * .9, a.dir);
        }
    }

    static void drawPoop(SceneCv c, double x, double y, double s, double pop, double T) {
        c.save(); c.translate(x, y); c.scale(s * pop, s * pop);
        c.fillStyle(rgba(0, 0, 0, .2)); c.beginPath(); c.ellipse(0, 1, 12, 3.4, 0, 0, PI2); c.fill();
        c.fillStyle("#6f431d"); c.beginPath(); c.ellipse(0, -3, 11, 6.4, 0, 0, PI2); c.fill();

    static void hud(SceneCore S, SceneCv c) {
        if (menuA != null) {
            double[] ctr = menuCenter(S);
            double el = S.t - menuT0;
            c.save();
            double rp = clamp(el / .35, 0, 1);
            c.strokeStyle(rgba(255, 255, 255, .55 * (1 - rp * .6))); c.lineWidth(2);
            c.beginPath(); c.arc(ctr[0], ctr[1], 10 + M_R * ease(rp) * 1.1, 0, PI2); c.stroke();
            for (int i = 0; i < 4; i++) {
                double p = clamp((el - i * .045) / .34, 0, 1);
                double e = p >= 1 ? 1 : 1 + 2.7 * Math.pow(p - 1, 3) + 1.7 * Math.pow(p - 1, 2);
                double px = ctr[0] + Math.cos(M_ANG[i]) * M_R * e, py = ctr[1] + Math.sin(M_ANG[i]) * M_R * e, r = M_IR * clamp(e, .05, 1.15);
                c.globalAlpha(clamp(p * 1.6, 0, 1));
                c.fillStyle(rgba(0, 0, 0, .18)); c.beginPath(); c.ellipse(px, py + 2.2, r, r * .9, 0, 0, PI2); c.fill();
                c.fillStyle(rgba(255, 255, 255, .96)); c.beginPath(); c.arc(px, py, r, 0, PI2); c.fill();
                c.strokeStyle(M_COL[i]); c.lineWidth(2.2); c.beginPath(); c.arc(px, py, r, 0, PI2); c.stroke();
                c.font(r * .95);
                c.fillStyle("#222222");
                c.fillTextC(M_ICON[i], px, py + 1);
                c.font(9);
                double tw = c.measureText(M_LABEL[i]) + 10;
                c.fillStyle(rgba(255, 255, 255, .94)); rr(c, px - tw / 2, py + r + 2, tw, 13, 6.5); c.fill();
                c.fillStyle(M_COL[i]); c.fillTextC(M_LABEL[i], px, py + r + 8.6);
            }
            c.globalAlpha(1);
            c.restore();
        }
    }

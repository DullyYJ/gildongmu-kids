    static final long HUNGER_STEP_MS = 12000L;          // 배고픔 1점이 오르는 시간
    static final long OFFLINE_CAP_MS = 20L * 60L * 1000L; // 앱을 닫아 둔 시간은 최대 이만큼만 센다
    static final int BEG_AT = 70;                       // 이 이상이면 밥 달라고 조른다
    static final long SULK_MS = 90000L;                 // 삐지는 시간
    static final long POOP_NEG_MS = 180000L;            // 응가 3개를 이만큼 두면 방치로 센다
    static final long NEG_DECAY_MS = 20L * 60L * 1000L; // 방치 횟수가 1씩 줄어드는 시간

    static final long CHAT_KEEP_MS = 7L * 24L * 3600L * 1000L; // 가족 대화 보관(최대 7일)
    static final long LOC_INTERVAL_MS = 300000L;        // 위치 보내는 주기(5분)
    static final long PAIR_TTL_MS = 10L * 60L * 1000L;  // 연결 비밀키 유효 시간
    static final int CHAT_MAX_LEN = 500;

    Engine(Store s) {
        store = s;
        String a = s.get("cid", "");
        if (a == null || a.isEmpty()) {
            a = UUID.randomUUID().toString();
            s.put("cid", a);
        }
        cid = a;
        String b = s.get("csec", "");
        if (b == null || b.isEmpty()) {
            byte[] r = new byte[16];
            srnd.nextBytes(r);
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < r.length; i++) sb.append(String.format(Locale.US, "%02x", r[i] & 0xff));
            b = sb.toString();
            s.put("csec", b);
        }
        csec = b;
        load();
    }

    // ───────── 저장·불러오기 ─────────
    private void load() {
        JSONObject o = new JSONObject();
        try {
            o = new JSONObject(store.get("state", "{}"));
        } catch (Exception e) {
            o = new JSONObject();
        }
        sel = o.optString("sel", CAST[0][0]);
        loc = o.optInt("loc", 0) == 1;
        lastFixT = o.optLong("lf", 0L);
        lastTickT = o.optLong("lt", 0L);
        pairCode = o.optString("pc", "");
        pairExp = o.optLong("pe", 0L);
        JSONArray a = o.optJSONArray("fr");
        for (int i = 0; i < CAST.length; i++) {
            F f = new F();
            f.id = CAST[i][0];
            f.name = CAST[i][1];
            f.kind = Integer.parseInt(CAST[i][2]);
            JSONObject s = null;
            if (a != null) {
                for (int j = 0; j < a.length(); j++) {
                    JSONObject x = a.optJSONObject(j);
                    if (x != null && f.id.equals(x.optString("i"))) s = x;
                }
            }
            if (s != null) {
                f.hunger = s.optDouble("h", 40);
                f.poop = s.optInt("p", 0);
                f.poopSince = s.optLong("ps", 0L);
                f.neg = s.optInt("n", 0);
                f.negT = s.optLong("nt", 0L);
                f.sick = s.optInt("sk", 0) == 1;
                f.starved = s.optInt("st", 0) == 1;
                f.poopNeg = s.optInt("pn", 0) == 1;
                f.sulkUntil = s.optLong("su", 0L);
            } else {
                f.hunger = 35 + rnd.nextInt(30);
            }
            fs.add(f);
        }
        try {
            chat = new JSONArray(store.get("chat", "[]"));
        } catch (Exception e) {
            chat = new JSONArray();
        }
    }

    private void save() {
        try {
            JSONObject o = new JSONObject();
            o.put("sel", sel);
            o.put("loc", loc ? 1 : 0);
            o.put("lf", lastFixT);
            o.put("lt", lastTickT);
            o.put("pc", pairCode);
            o.put("pe", pairExp);
            JSONArray a = new JSONArray();
            for (F f : fs) {
                JSONObject x = new JSONObject();
                x.put("i", f.id);
                x.put("h", f.hunger);
                x.put("p", f.poop);
                x.put("ps", f.poopSince);
                x.put("n", f.neg);
                x.put("nt", f.negT);
                x.put("sk", f.sick ? 1 : 0);
                x.put("st", f.starved ? 1 : 0);
                x.put("pn", f.poopNeg ? 1 : 0);
                x.put("su", f.sulkUntil);
                a.put(x);
            }
            o.put("fr", a);
            store.put("state", o.toString());
            store.put("chat", chat.toString());
        } catch (Exception e) {
            // 다음 기회에 다시 저장
        }
        dirty = false;
        lastSave = System.currentTimeMillis();
    }

    // ───────── 시계 한 틱 ─────────
    synchronized void tick(long now) {
        lastNow = now;
        long dt = lastTickT == 0L ? 0L : now - lastTickT;
        if (dt < 0L) dt = 0L;
        if (dt > OFFLINE_CAP_MS) dt = OFFLINE_CAP_MS;
        lastTickT = now;
        for (int i = 0; i < fs.size(); i++) step(fs.get(i), now, dt);
        if (now - lastPrune > 60000L) {
            lastPrune = now;
            pruneChat(now);
        }
        if (pairExp != 0L && now > pairExp) {
            pairCode = "";
            pairExp = 0L;
        }
        String sg = sig();
        if (!sg.equals(lastSig)) {
            lastSig = sg;
            rev++;
        }
        dirty = true;
        if (now - lastSave > 5000L) save();
    }

    private void step(F f, long now, long dt) {
        if (!f.sick && f.hunger < 100) f.hunger = Math.min(100.0, f.hunger + dt / (double) HUNGER_STEP_MS);
        if (f.hunger >= 100 && !f.starved) {
            f.starved = true;
            neglect(f, now);
        }
        if (f.poop >= 3 && !f.poopNeg && f.poopSince > 0 && now - f.poopSince > POOP_NEG_MS) {
            f.poopNeg = true;
            neglect(f, now);
        }
        if (f.neg > 0 && !f.sick && f.negT > 0 && now - f.negT > NEG_DECAY_MS) {
            f.neg--;
            f.negT = now;
        }
        if (f.tmUntil != 0L && now >= f.tmUntil) {
            f.tm = "";
            f.tmUntil = 0L;
        }
        if (f.nextEvt == 0L) f.nextEvt = now + EVT_MIN + rnd.nextInt((int) (EVT_MAX - EVT_MIN));
        if (now >= f.nextEvt) {
            f.nextEvt = now + EVT_MIN + rnd.nextInt((int) (EVT_MAX - EVT_MIN));
            if (!f.sick && f.sulkUntil <= now && f.tm.isEmpty()) event(f, now);
        }
    }

    /** 돌봄을 놓친 횟수를 센다 — 1·2번째는 삐지고, 3번째부터는 아프다(죽지는 않는다). */
    private void neglect(F f, long now) {
        f.neg++;
        f.negT = now;
        if (f.neg >= 3) {
            f.sick = true;
            f.sulkUntil = 0L;
        } else {
            f.sulkUntil = now + SULK_MS;
        }
    }

    private void event(F f, long now) {
        int r = rnd.nextInt(100);
        if (r < 35) {
            if (f.poop < 3) {
                if (f.poop == 0) f.poopSince = now;
                f.poop++;
            }
            setTm(f, "poop", now + 4000L);
        } else if (r < 65) {
            setTm(f, "play", now + 6000L);
        } else if (r < 85) {
            setTm(f, "sleep", now + 10000L);
        }
    }

    private void setTm(F f, String m, long until) {
        f.tm = m;
        f.tmUntil = until;
    }

    private int moodOf(F f, long now) {
        if (f.sick) return 4;
        if (f.sulkUntil > now) return 3;
        if (!f.tm.isEmpty()) {
            for (int i = 0; i < MOODS.length; i++) if (MOODS[i].equals(f.tm)) return i;
        }
        if (f.hunger >= BEG_AT) return 1;
        return 0;
    }

    private F find(String id) {
        for (int i = 0; i < fs.size(); i++) if (fs.get(i).id.equals(id)) return fs.get(i);
        return fs.get(0);
    }

    private String sig() {
        StringBuilder sb = new StringBuilder();
        sb.append(tab).append('|').append(theme()).append('|').append(consented() ? 1 : 0)
                .append(store.get("cp", "0")).append(store.get("cc", "0")).append(loc ? 1 : 0)
                .append('|').append(sel).append('|').append(pairCode).append('|').append(chat.length());
        if ("settings".equals(tab)) sb.append('|').append(lastFixT / 60000L);
        for (int i = 0; i < fs.size(); i++) {
            F f = fs.get(i);
            sb.append('|').append(moodOf(f, lastNow)).append(',').append((int) (f.hunger / 5)).append(',')
                    .append(f.poop).append(',').append(f.neg);
        }
        return sb.toString();
    }

    synchronized String village() {
        try {
            JSONObject o = new JSONObject();
            o.put("sel", sel);
            JSONArray a = new JSONArray();
            for (int i = 0; i < fs.size(); i++) {
                F f = fs.get(i);
                int m = moodOf(f, lastNow);
                int cell = CELLS[f.kind][m];
                String mode;
                if (m == 0) mode = "walk";
                else if (cell >= 0) mode = "cell";
                else if (m == 3 || m == 4 || m == 7) mode = "still";
                else mode = "walk";
                JSONObject x = new JSONObject();
                x.put("id", f.id);
                x.put("name", f.name);
                x.put("mode", mode);
                x.put("cell", cell);
                x.put("bubble", BUBBLE[m]);
                x.put("poop", f.poop);
                a.put(x);
            }
            o.put("friends", a);
            return o.toString();
        } catch (Exception e) {
            return "{}";
        }
    }

    static boolean batchim(String s) {
        if (s == null || s.isEmpty()) return false;
        char c = s.charAt(s.length() - 1);
        if (c < 0xAC00 || c > 0xD7A3) return false;
        return (c - 0xAC00) % 28 != 0;
    }

    static String ga(String n) {
        return n + (batchim(n) ? "이" : "가");
    }

    private int need(F f, long now) { // 0 아픔 · 1 삐짐 · 2 배고픔 · 3 응가 · 4 괜찮음
        if (f.sick) return 0;
        if (f.sulkUntil > now) return 1;
        if (f.hunger >= BEG_AT) return 2;
        if (f.poop > 0) return 3;
        return 4;
    }

    private String line(F f, long now) {
        switch (need(f, now)) {
            case 0:
                return ga(f.name) + " 아파요… 밥을 주고 응가를 치워 주면 나아져요";
            case 1:
                return ga(f.name) + " 삐졌어요. 쓰다듬어 주세요";
            case 2:
                return ga(f.name) + " 배고파요 밥을 주세요";
            case 3:
                return ga(f.name) + " 응가했어요 치워 주세요";
            default:
                return ga(f.name) + " 잘 지내요";
        }
    }

        } else if (v.equals("consent.parent")) {

        } else if (v.equals("consent.child")) {
            store.put("cc", "1".equals(store.get("cc", "0")) ? "0" : "1");
        } else if (v.equals("consent.ok")) {
            if ("1".equals(store.get("cp", "0")) && "1".equals(store.get("cc", "0"))) {
                store.put("consent", "1");
                loc = true;
                tab = "home";
                res = cmd("perm");
            }
        } else if (v.equals("consent.reset")) {
            store.put("consent", "0");
            store.put("cp", "0");
            store.put("cc", "0");
            loc = false;
            tab = "home";
            res = cmd("loc.stop");
        } else if (v.equals("loc.toggle")) {
            loc = !loc;
            res = loc ? cmd("perm") : cmd("loc.stop");
        } else if (v.equals("loc.denied")) {
            loc = false;
        } else if (v.equals("pair.new")) {
            pairCode = genKey();
            pairExp = now + PAIR_TTL_MS;
        } else if (v.equals("chat.send")) {
            String s = arg == null ? "" : arg.trim();
            if (s.length() > CHAT_MAX_LEN) s = s.substring(0, CHAT_MAX_LEN);
            if (!s.isEmpty()) {
                try {
                    JSONObject m = new JSONObject();
                    m.put("t", now);
                    m.put("m", 1);
                    m.put("x", s);
                    chat.put(m);
                } catch (Exception e) {
                    // 무시
                }
            }
        } else if (v.startsWith("sel:")) {
            sel = find(v.substring(4)).id;
        } else if (v.startsWith("friend:")) {
            sel = find(v.substring(7)).id;
            tab = "friends";
        } else if (v.startsWith("clean:")) {
            F f = find(v.substring(6));
            if (f.poop > 0) f.poop--;
            if (f.poop == 0) {
                f.poopNeg = false;
                f.poopSince = 0L;
            }
            recover(f, now);
        } else if (v.equals("care.feed")) {
            F f = find(sel);
            f.hunger = Math.max(0.0, f.hunger - 50.0);
            if (f.hunger < 100) f.starved = false;
            if (f.sulkUntil > now) f.sulkUntil = 0L;
            if (f.sick) recover(f, now);
            if (!f.sick) setTm(f, "eat", now + 4000L);
        } else if (v.equals("care.pet")) {
            F f = find(sel);
            f.sulkUntil = 0L;
            if (f.sick) recover(f, now);
            if (!f.sick) setTm(f, "play", now + 3000L);
        } else if (v.equals("care.clean")) {
            F f = find(sel);
            f.poop = 0;
            f.poopNeg = false;
            f.poopSince = 0L;
            recover(f, now);
        }
        dirty = true;
        save();
        return res;
    }

            }

    private void move(A x, float dt) {

        if (x.wait > 0f) {
            x.wait -= dt;

            x.wait = 0.6f + rnd.nextFloat() * 3f;
            x.tx = 0.08f + rnd.nextFloat() * 0.84f;
            return;
        }
        x.dir = d > 0f ? 1f : -1f;
        float sp = 0.07f * dt;
        x.x += Math.abs(d) < sp ? d : x.dir * sp;
    }

    // ───────── 화면 한 장 ─────────
    void build(String json) {
        JSONObject o;
        try {
            o = new JSONObject(json);
        } catch (Exception e) {
            return;
        }
        host.setBackgroundColor(Pal.c(Pal.BG));
        loading.setTextColor(Pal.c(Pal.SUB));
        if ("map".equals(o.optString("mode"))) {
            showMap();
            return;
        }
        if (map != null) map.setVisibility(View.GONE);
        loading.setVisibility(View.GONE);
        scroll.setVisibility(View.VISIBLE);
        scroll.setBackgroundColor(Pal.c(Pal.BG));
        JSONArray bl = o.optJSONArray("blocks");
        String head = "";
        if (bl != null && bl.length() > 0 && bl.optJSONObject(0) != null) head = bl.optJSONObject(0).optString("text");
        final boolean same = head.equals(lastHead);
        lastHead = head;
        final int sy = same ? scroll.getScrollY() : 0;
        final boolean bottom = "1".equals(o.optString("bottom"));
        col.removeAllViews();
        if (bl != null) {
            for (int i = 0; i < bl.length(); i++) {
                JSONObject b = bl.optJSONObject(i);
                if (b == null) continue;
                View v = block(b);
                if (v != null) col.addView(v);
            }
        }
        scroll.post(new Runnable() {
            @Override
            public void run() {
                if (bottom) scroll.fullScroll(View.FOCUS_DOWN);
                else scroll.scrollTo(0, sy);
            }
        });
    }

    private void showMap() {
        scroll.setVisibility(View.GONE);
        if (map != null) {
            loading.setVisibility(View.GONE);
            map.setVisibility(View.VISIBLE);
            return;
        }
        loading.setText("지도를 불러오는 중이에요…");
        loading.setVisibility(View.VISIBLE);
        if (mapLoading) return;
        mapLoading = true;
        new Thread(new Runnable() {
            @Override
            public void run() {
                MetroMapData d = null;
                try {
                    InputStream in = ctx.getAssets().open("metro_native.bin");
                    try {
                        d = MetroMapData.parse(in);
                    } finally {
                        in.close();
                    }
                } catch (Throwable t) {
                    d = null;
                }
                final MetroMapData data = d;
                host.post(new Runnable() {
                    @Override
                    public void run() {
                        mapReady(data);
                    }
                });
            }
        }, "kids-map").start();
    }

    private void mapReady(MetroMapData data) {
        mapLoading = false;
        if (data == null) {
            loading.setText("지도를 불러오지 못했어요");
            return;
        }
        map = new MetroMapView(ctx, data, ctx.getResources().getDisplayMetrics().density);
        map.setListener(new MetroMapView.Listener() {
            @Override
            public void onStation(int op, float sx, float sy) {
                // 길찾기는 쓰지 않는다 — 역을 눌러도 아무 일도 없다
            }

            @Override
            public void onGestureStart() {
            }

            @Override
            public void onEdgeSwipe(int dir) {
            }

            @Override
            public void onReady() {
            }

            @Override
            public void onError(String msg) {
            }

            @Override
            public void onCentered(int op) {
            }
        });
        map.setVisibility(View.GONE);
        host.addView(map, 1, new FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT));
        if (refresh != null) refresh.run();
    }

    private Bitmap qr(String text) {
        try {
            BitMatrix m = new QRCodeWriter().encode(text, BarcodeFormat.QR_CODE, 0, 0);
            int n = m.getWidth(), sc = 8, w = n * sc;
            int[] px = new int[w * w];
            for (int y = 0; y < w; y++) {
                for (int x = 0; x < w; x++) px[y * w + x] = m.get(x / sc, y / sc) ? 0xFF000000 : 0xFFFFFFFF;
            }
            return Bitmap.createBitmap(px, w, w, Bitmap.Config.ARGB_8888);
        } catch (Throwable t) {
            return null;
        }
    }

    private static final class Bar extends View {
        private final int val;
        private final Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
        private final RectF r = new RectF();

        Bar(Context c, int val) {
            super(c);
            this.val = val;
        }

        @Override
        protected void onDraw(Canvas c) {
            float w = getWidth(), h = getHeight();
            p.setColor(Pal.c(Pal.BORDER));
            r.set(0, 0, w, h);
            c.drawRoundRect(r, h / 2f, h / 2f, p);
            if (val > 0) {
                p.setColor(val >= 70 ? Pal.c(Pal.ORANGE) : Pal.c(Pal.GOLD));
                r.set(0, 0, Math.max(h, w * val / 100f), h);
                c.drawRoundRect(r, h / 2f, h / 2f, p);
            }
        }
    }

    private static final class Toggle extends View {
        private final boolean on;
        private final Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
        private final RectF r = new RectF();

        Toggle(Context c, boolean on) {
            super(c);
            this.on = on;
        }

        @Override
        protected void onDraw(Canvas c) {
            float w = getWidth(), h = getHeight();
            p.setColor(on ? Pal.c(Pal.GOLD) : Pal.c(Pal.BORDER2));
            r.set(0, 0, w, h);
            c.drawRoundRect(r, h / 2f, h / 2f, p);
            p.setColor(0xFFFFFFFF);
            float rad = h / 2f - h * 0.12f;
            c.drawCircle(on ? w - h / 2f : h / 2f, h / 2f, rad, p);
        }
    }

    }

    // ───────── 엔진 ↔ 화면 ─────────
    /** 엔진 스레드에서: 한 틱 돌리고, 바뀐 게 있으면 새 화면을 그리게 넘긴다. */
    private void step() {
        Engine en = Core.engine;
        en.tick(System.currentTimeMillis());
        final String theme = en.theme();
        final String tab = en.tab();
        int rev = en.rev();
        final String vil = en.village();
        if (rev != lastRev || !tab.equals(lastTab) || !theme.equals(lastTheme)) {
            lastRev = rev;
            lastTab = tab;
            lastTheme = theme;
            final String scr = en.screen(tab);
            final String bar = en.tabbar(tab);
            ui.post(new Runnable() {
                @Override
                public void run() {
                    show(theme, scr, bar, vil);
                }
            });
        } else {
            ui.post(new Runnable() {
                @Override
                public void run() {
                    renderer.village(vil);
                }
            });
        }
    }

    private void show(String theme, String scr, String bar, String vil) {
        Pal.set(theme);
        applyBars();
        root.setBackgroundColor(Pal.c(Pal.BG));
        renderer.build(scr);
        renderer.village(vil);
        tabbar.set(bar, new TabBar.Cb() {
            @Override
            public void tab(String id) {
                act("tab." + id, "");
            }
        });
        tabbar.setVisibility(tabbar.count() > 0 ? View.VISIBLE : View.GONE);
    }

    private void applyBars() {
        Window w = getWindow();
        w.setStatusBarColor(Pal.c(Pal.BG));
        w.setNavigationBarColor(Pal.c(Pal.NAV));
        WindowInsetsControllerCompat ic = WindowCompat.getInsetsController(w, w.getDecorView());
        ic.setAppearanceLightStatusBars(!Pal.dark);
        ic.setAppearanceLightNavigationBars(!Pal.dark);
    }

    /** 눌림은 그대로 엔진에 넘긴다. 결과로 나온 명령(위치 권한·시작·멈춤)만 여기서 한다. */
    private void act(final String v, final String arg) {
        eh.post(new Runnable() {
            @Override
            public void run() {
                String res = Core.engine.call(v, arg, System.currentTimeMillis());
                handle(res);
                step();
            }
        });
    }

    private void handle(String res) {
        if (res == null || res.isEmpty()) return;
        String cmd = "";
        try {
            cmd = new JSONObject(res).optString("cmd");
        } catch (Exception e) {
            cmd = "";
        }
        if ("perm".equals(cmd)) {
            ui.post(new Runnable() {
                @Override
                public void run() {
                    askPerms();
                }
            });
        } else if ("loc.start".equals(cmd)) {
            ui.post(new Runnable() {
                @Override
                public void run() {
                    startLoc();
                }
            });
        } else if ("loc.stop".equals(cmd)) {
            ui.post(new Runnable() {
                @Override
                public void run() {
                    stopLoc();
                }
            });
        }
    }

    // ───────── 위치 권한·서비스 ─────────
    private boolean hasLocPerm() {
        return checkSelfPermission(Manifest.permission.ACCESS_FINE_LOCATION) == PackageManager.PERMISSION_GRANTED;
    }

    private void askPerms() {
        ArrayList<String> need = new ArrayList<String>();
        if (!hasLocPerm()) {
            need.add(Manifest.permission.ACCESS_FINE_LOCATION);
            need.add(Manifest.permission.ACCESS_COARSE_LOCATION);
        }
        if (Build.VERSION.SDK_INT >= 33
                && checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED) {
            need.add(Manifest.permission.POST_NOTIFICATIONS);
        }
        if (need.isEmpty()) {
            if (Core.engine.locOn()) startLoc();
            return;
        }
        requestPermissions(need.toArray(new String[0]), REQ_PERM);
    }

    @Override
    public void onRequestPermissionsResult(int code, String[] perms, int[] results) {
        super.onRequestPermissionsResult(code, perms, results);
        if (code != REQ_PERM) return;
        if (hasLocPerm()) {
            if (Core.engine.locOn()) startLoc();
        } else {
            act("loc.denied", "");
        }
    }

    private void startLoc() {
        if (!hasLocPerm()) return;
        try {
            startForegroundService(new Intent(this, LocationService.class));
        } catch (Throwable t) {
            // 서비스를 못 켜도 앱은 계속 쓴다
        }
    }

    private void stopLoc() {
        stopService(new Intent(this, LocationService.class));
    }
}
